<script>
  let vocabDB = [];
  let idiomsDB = [];
  let tensesDB = [];

  const quotes = [
    '🔥 "Every word you learn gets you closer to C1 & C2!"',
    '⭐ "Great things take time. Keep pushing your limits, Abdulrahman!"',
    '🎯 "Consistency is the key to mastering English syntax & vocab."',
    '🚀 "Your future self will thank you for studying today!"',
    '💪 "Small daily progress leads to massive C2 results."'
  ];

  let quoteIdx = 0;
  setInterval(() => {
    quoteIdx = (quoteIdx + 1) % quotes.length;
    document.getElementById('motivationQuote').innerText = quotes[quoteIdx];
  }, 4000);

  function addVocab() {
    const word = document.getElementById('vocabWord').value;
    const meaning = document.getElementById('vocabMeaning').value;
    const sentence = document.getElementById('vocabSentence').value;
    if (!word || !meaning) return;

    vocabDB.unshift({ word, meaning, sentence });
    document.getElementById('vocabWord').value = '';
    document.getElementById('vocabMeaning').value = '';
    document.getElementById('vocabSentence').value = '';
    renderVocabBox();
  }

  function renderVocabBox() {
    const box = document.getElementById('vocabSavedList');
    box.innerHTML = '';
    vocabDB.forEach(item => {
      box.innerHTML += `<div class="item-card"><div class="flex-row"><span class="ar-text" style="font-weight:bold;">${item.meaning}</span><span class="en-text" style="color:var(--accent-purple); font-weight:bold;">${item.word}</span></div></div>`;
    });
  }

  function addIdiom() {
    const text = document.getElementById('idiomText').value;
    const meaning = document.getElementById('idiomMeaning').value;
    if (!text || !meaning) return;

    idiomsDB.unshift({ text, meaning });
    document.getElementById('idiomText').value = '';
    document.getElementById('idiomMeaning').value = '';
    renderIdiomBox();
  }

  function renderIdiomBox() {
    const box = document.getElementById('idiomSavedList');
    box.innerHTML = '';
    idiomsDB.forEach(item => {
      box.innerHTML += `<div class="item-card"><div class="flex-row"><span class="ar-text">${item.meaning}</span><span class="en-text" style="color:var(--accent-amber); font-weight:bold;">${item.text}</span></div></div>`;
    });
  }

  function addTense() {
    const tense = document.getElementById('tenseSelect').value;
    const ex = document.getElementById('tenseExample').value;
    if (!ex) return;

    tensesDB.unshift({ tense, ex });
    document.getElementById('tenseExample').value = '';
    renderTenseBox();
  }

  function renderTenseBox() {
    const box = document.getElementById('tenseSavedList');
    box.innerHTML = '';
    tensesDB.forEach(item => {
      box.innerHTML += `<div class="item-card"><div style="color:var(--accent-rose); font-weight:bold; font-size:9.5pt;">${item.tense}</div><div class="en-text">${item.ex}</div></div>`;
    });
  }

  function checkAns(btn, isCorrect) {
    btn.style.background = isCorrect ? '#22c55e' : '#ef4444';
    btn.style.borderColor = isCorrect ? '#16a34a' : '#dc2626';
  }

  function startVocabPractice() {
    const area = document.getElementById('vocabPracticeArea');
    area.innerHTML = '';
    if (vocabDB.length === 0) { area.innerHTML = '<p style="color:var(--accent-rose); font-weight:bold; margin-top:10px;">أضف كلمات أولاً لتظهر في التمارين!</p>'; return; }
    
    const q = vocabDB[0];
    area.innerHTML = `
      <div class="item-card" style="margin-top:10px;">
        <div style="font-weight:bold; text-align:center; font-size:14pt;" class="en-text">${q.word}</div>
        <div class="grid-2x2">
          <button class="opt-btn" onclick="checkAns(this, true)">${q.meaning}</button>
          <button class="opt-btn" onclick="checkAns(this, false)">تفاح 🍎</button>
          <button class="opt-btn" onclick="checkAns(this, false)">سيارة 🚗</button>
          <button class="opt-btn" onclick="checkAns(this, false)">كتاب 📚</button>
        </div>
      </div>
    `;
  }

  function startIdiomPractice() {
    const area = document.getElementById('idiomPracticeArea');
    area.innerHTML = '';
    if (idiomsDB.length === 0) { area.innerHTML = '<p style="color:var(--accent-rose); font-weight:bold; margin-top:10px;">أضف مصطلحات أولاً!</p>'; return; }
    
    const item = idiomsDB[0];
    area.innerHTML = `
      <div class="item-card" style="margin-top:10px;">
        <div style="text-align:center; font-weight:bold;" class="ar-text">${item.meaning}</div>
        <div class="grid-2x2">
          <button class="opt-btn" onclick="checkAns(this, true)">${item.text}</button>
          <button class="opt-btn" onclick="checkAns(this, false)">Break a leg 🎭</button>
          <button class="opt-btn" onclick="checkAns(this, false)">Under the weather 🌧️</button>
          <button class="opt-btn" onclick="checkAns(this, false)">Spill the beans 🫘</button>
        </div>
      </div>
    `;
  }

  function startTensePractice() {
    const area = document.getElementById('tensePracticeArea');
    area.innerHTML = `
      <div class="item-card" style="margin-top:10px;">
        <p style="margin:0 0 8px 0; font-size:10pt;">اختر الزمن المناسب للجملة:</p>
        <div class="en-text" style="margin-bottom:12px; font-weight:bold; color:var(--accent-blue);">"I have been studying English for 2 hours."</div>
        <div class="grid-2x2">
          <button class="opt-btn" onclick="checkAns(this, true)">Present Perfect</button>
          <button class="opt-btn" onclick="checkAns(this, false)">Past Simple</button>
          <button class="opt-btn" onclick="checkAns(this, false)">Present Simple</button>
          <button class="opt-btn" onclick="checkAns(this, false)">Future Simple</button>
        </div>
      </div>
    `;
  }
</script>
